constexpr double kRecoilTime[] = {0.0, 0.1, 0.3, 0.5, 0.7, 0.9, 1.1};
constexpr double kRecoilPath[] = {0.0, 0.2, 1.3, 4.3, 7.7, 10.7, 12.6};
double recoilPath(double t)
{
	constexpr int n = 7;
	if (t <= 0.0)
		return 0.0;
	for (int i = 1; i < n; ++i)
	{
		if (t <= kRecoilTime[i])
			return kRecoilPath[i - 1] + (kRecoilPath[i] - kRecoilPath[i - 1]) * (t - kRecoilTime[i - 1]) / (kRecoilTime[i] - kRecoilTime[i - 1]);
	}
	return kRecoilPath[n - 1];
}
