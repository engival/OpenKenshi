	ecs.each([&](const state::Id& instance, state::Door& door)
	{
		if (door.state != 1)
			return;
		for (std::size_t i = 0; i < members.size(); ++i)
		{
			const state::Fighter& walker = members[i].fighter;
			if (walker.dead || walker.unconscious || !walker.walking())
				continue;
			const state::InteractTask* task = ecs.entity(combatEntities_[i]).get<state::InteractTask>();
			if (task != nullptr && task->target == instance.value)
				continue;
			if (std::hypot(walker.x - door.x, walker.z - door.z) <= kDoorOpenReach && std::fabs(walker.y - door.y) <= kInteractReachY)
			{
				door.state = 0;
				++door.version;
				return;
			}
		}
	});
