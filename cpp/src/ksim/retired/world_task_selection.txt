// The leader of a built world squad (residents.cpp's rule): its first member in Id order that is alive, conscious, not the player's and
// on no Routine of its own.
std::uint64_t leaderOf(const CombatArena& combat, const std::vector<std::pair<std::uint64_t, flecs::entity>>& members)
{
	for (const auto& [id, member] : members)
	{
		const state::Fighter* f = combat.find(id);
		if (f != nullptr && !f->dead && !f->unconscious && !f->playerControlled && !member.has<state::Routine>())
			return id;
	}
	return 0;
}

// Whether any AI package the template links carries `task`. The live caravan's chosen package is the wandering one, but its deliver
// package (192) is a second link; the trigger is "the package goal", so any link counts. The town scorer reads only the registry's
// chosen package (some squad templates map to a package family the registry resolves elsewhere).
bool templateCarries(const World& world, const routines::PackageTable& packages, const std::string& templateSid, std::uint16_t task)
{
	const auto prefab = world.prefabs().find(templateSid);
	if (prefab == world.prefabs().end() || !prefab->second.has<tmpl::SquadTemplate>())
		return false;
	bool carries = false;
	prefab->second.each<tmpl::AiPackageLink>([&](flecs::entity package)
	{
		const tmpl::Source* source = package.get<tmpl::Source>();
		const routines::CompiledPackage* compiled = source != nullptr ? packages.find(source->stringID) : nullptr;
		carries = carries || (compiled != nullptr && routines::carriesGoal(*compiled, task));
	});
	return carries;
}
