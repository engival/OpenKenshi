The rule, as it stood:

// Whether the two factions are hostile: either one's relation to the other is at or below FactionBook::kHostile.
bool hostileEither(const kdata::GameData& data, flecs::entity a, flecs::entity b)
{
	return factionRelation(data, a, b) <= FactionBook::kHostile || factionRelation(data, b, a) <= FactionBook::kHostile;
}
