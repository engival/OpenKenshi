DoorDecision planDoorAccess(const DoorFacts& facts) noexcept
{
	if (facts.mine)
		return {DoorPlan::Unlock, 1.0, false};
	// A door nobody owns has no faction to score: its kind counts as none (0).
	const bool bashable = !facts.allied && (facts.hostile || (facts.hasOwner && kindBashesDoors(facts.ownerKind)));
	DoorDecision best;
	if (bashable)
	{
		if (facts.weaponInHand || facts.weaponCarried)
			best = {DoorPlan::Bash, kBashUtility, !facts.weaponInHand};
		if (facts.gate && facts.mates < kGateCrewCap && !facts.underAttack)
		{
			const double utility = kBashUtility / static_cast<double>(facts.mates + 1);
			if (utility > best.utility)
				best = {DoorPlan::Bash, utility, false};
		}
	}
	// PICK_LOCK needs empty hands: a carried weapon does not count against it, a wielded one does.
	if (!facts.weaponInHand && facts.lockLevel - facts.lockpicking < kPickMaxDifficulty && kPickUtility > best.utility)
		best = {DoorPlan::Pick, kPickUtility, false};
	return best;
}
